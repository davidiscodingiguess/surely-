The service-worker contract (Phase 5) 
self.addEventListener('notificationclick', e => {
  const d = e.notification.data; // {approveUrl, declineUrl, deepLink}
  e.notification.close();
  if (e.notification.tag) {/* re-push replaced in place */}
  if (e.action === 'approve' || e.action === 'decline') {
    // headless: no window opens
    e.waitUntil(fetch(d[e.action + 'Url'],
      {method:'POST', credentials:'include'}));
  } else {
    // body tap → focus or open the PWA at the card
    e.waitUntil(clients.matchAll({type:'window'})
      .then(ws => ws.length
        ? ws[0].focus().then(c => c.navigate(d.deepLink))
        : clients.openWindow(d.deepLink)));
  }
});
